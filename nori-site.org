(defun nori-site-build-entry-index (&optional org-file)
  "Scan the ORG-FILE buffer - source file holding posts - and build a global index of metadata and contents.
Stores the result in 'nori-site-post-index'."
  (interactive)
  (let ((o-buffer
	 (if org-file (find-file-noselect org-file)
	   (current-buffer)))
	(index '()))
    (with-current-buffer o-buffer
      ;; Parse the buffer once
      (let ((ast (org-element-parse-buffer)))
	;; Process each headline
	(org-element-map ast 'headline
	  (lambda (headline)
	    (let* ((begin (org-element-property :begin headline))
		   (props (org-entry-properties begin 'all))
		   (file-name (alist-get
			       "EXPORT_FILE_NAME"
			       props nil nil #'string=))
		   (section-name (alist-get
				  "EXPORT_SECTION"egin
				  props nil nil #'string=)))
	      ;; Only process sections and files
	      (when (or section-name file-name)
		;; Extract everything
		(let* ((contents-begin (org-element-property :contents-begin headline))
		       (contents-end (org-element-property :contents-end headline))
		       (level (org-element-property :level headline))
		       (ancestors 
			(org-element-lineage-map headline
			    (lambda (parent)
			      (when (eq (org-element-type parent) 'headline)
				parent))))
		       (ancestors-begin (mapcar (lambda (parent)
						  (org-element-property :begin parent))
						ancestors))
		       (ancestors-props (mapcar (lambda (an-begin)
						  (org-entry-properties an-begin 'all))
						ancestors-begin))
		       (ancestors-names (mapcar (lambda (parent)
						  (substring-no-properties
						   (org-element-interpret-data
						    (org-element-property :title parent))))
						ancestors))
		       (path-chain (mapcar (lambda (an-props)
					     (alist-get "EXPORT_SECTION" an-props nil nil #'string=))
					   ancestors-props))

		       (title (substring-no-properties
			       (org-element-interpret-data
				(org-element-property :title headline))))
		       (date (alist-get
			      "EXPORT_DATE"
			      props nil nil #'string=))
		       (categories (org-entry-get
				    begin
				    "EXPORT_CATEGORIES"
				    :inherit))
		       (layout (alist-get "EXPORT_LAYOUT"
					  props nil nil #'string=))
		       (f-image (alist-get "EXPORT_FEATURED_IMAGE"
					   props nil nil #'string=))
		       (f-alt (alist-get "EXPORT_FEATURED_ALT"
					 props nil nil #'string=))
		       (description (alist-get "DESCRIPTION"
					       props nil nil #'string=))
		       (menu-name (alist-get "MENU_NAME"
					     props nil nil #'string=))
		       (todo (org-entry-get
			      begin
			      "TODO"
			      :inherit))
		       (tags (org-entry-get
			      begin
			      "ALLTAGS"
			      :inherit))
		       (publish-dir (expand-file-name nori-site-publish-directory))
		       (target-dir publish-dir)
		       (final-filename nil)
		       (url
			(if (or (string= file-name "_index")(string= file-name "index"))
			    (concat "/" (string-join path-chain "/"))
			  (concat "/" (string-join path-chain "/") "/" file-name)))
		       (contents
			(when contents-begin ;just in case it's empty
			  (buffer-substring contents-begin contents-end)))
		       (wordcount (count-words contents-begin contents-end)))

		  
		  ;; Determine target directory and final-filename
		  ;; Append path chain
		  (dolist (part (nreverse path-chain))
		    (setq target-dir (expand-file-name part target-dir)))
		  ;; Append current section name if exists
		  (when section-name
		    (setq target-dir (expand-file-name section-name target-dir)))
		  
		  ;; Determine filename
		  (if file-name
		      (if (or (string= file-name "_index")(string= file-name "index"))
			  ;; Index files go directly in the calculated target-dir
			  (setq final-filename (expand-file-name "index.html" target-dir))
			;; Regular posts get their own subfolder /target/post-name/index.html
			(setq target-dir (expand-file-name file-name target-dir))
			(setq final-filename (expand-file-name "index.html" target-dir))))		  

		  ;; Build the index entry for the headline
		  (let ((meta `((begin . ,begin)
				(file-name . ,file-name)
				(section-name . ,section-name)
				(contents-begin . ,contents-begin)
				(contents-end . ,contents-end)
				(level . ,level)
				(path-chain . ,(nreverse path-chain))
				(ancestors-names . ,(nreverse ancestors-names))
				(title . ,title)
				(date . ,date)
				(categories . ,categories)
				(layout . ,layout)
				(f-image . ,f-image)
				(f-alt . ,f-alt)
				(description . ,description)
				(menu-name . ,menu-name)
				(todo . ,todo)
				(tags . ,tags)
				(target-dir . ,target-dir)
				(final-filename . ,final-filename)
				(url . ,url)
				(wordcount . ,wordcount)
					;(contents . ,contents)
				)))
		    ;; Add to index
		    (push (cons title meta) index)))))))))
    ;; Store globally
    (setq nori-site-entry-index index)
    (message "Indexed %d posts." (length index))
    index
    ))

(defun nori-site-render-all-pages ()
  "Iterate over 'nori-site-entry-index', which contains all the entries parsed from org-file, and render each page."
  (interactive)
  ;; Parse the org-source if not parsed yet
  (unless nori-site-entry-index
    (nori-site-build-entry-index nori-site-source-org-filename))

  ;; Run the css compilation and minification
  (shell-command "bash /home/nori/Documents/Notes/website/process-css.sh")

  (let* ((publish-dir (expand-file-name nori-site-publish-directory))
	(org-buffer (find-file-noselect nori-site-source-org-filename))
	(posts (nori-site-gather-items "post"))
	(poems (nori-site-gather-items "poem"))
	(archive-html (nori-site-generate-archive-sidebar posts)))

    ;; Ensure publish directory exists
    (unless (file-exists-p publish-dir)
      (make-directory publish-dir t))
    
    ;; Copy assets to publish-dir
    (copy-directory nori-site-content-img-dir publish-dir t t nil)
    (copy-directory nori-site-img-dir (concat publish-dir "img/") t t t) ; args are keep-time parents and copy-contents
    (copy-directory nori-site-css-dir publish-dir t t nil)
    (copy-directory nori-site-fonts-dir publish-dir t t nil)
    
    (message "Index length %d" (length nori-site-entry-index))
    
  ;; Loop through index to render
  (dolist (entry nori-site-entry-index)
    
    (let* ((title (car entry))
	   (meta (cdr entry))
	   (section-name (alist-get 'section-name meta))
	   (file-name (alist-get 'file-name meta))
	   (begin (alist-get 'begin meta))
	   (todo-state (alist-get 'todo meta))
	   (target-dir (alist-get 'target-dir meta))
	   (final-filename (alist-get 'final-filename meta)))

      ;; Skip TODO items
      (unless (and todo-state (string= todo-state "TODO"))
	;; Ensure directory exists
	(unless (file-exists-p target-dir)
	  (make-directory target-dir t))

	;; Render
	(when final-filename
	  (let* ((html (nori-site-export-subtree-to-html org-buffer begin))
		 (page (nori-site-generate-page meta html archive-html posts poems))
	      	 )
	    (with-temp-file final-filename
	      (insert page))
	    )
	  (message "Published: %s -> %s" title file-name))
	)))))
(nori-site-render-all-pages)

(defun nori-site-generate-page (metadata html archive-html posts poems)
  "Generate a page using the template."
  (let* ((layout (alist-get 'layout metadata))
	 (title (alist-get 'title metadata))
	 (file-name (alist-get 'file-name metadata))
	 (url (alist-get 'url metadata))
	 (section-name (alist-get 'section-name metadata))
	 (f-image (alist-get 'f-image metadata))
	 (f-alt (alist-get 'f-alt metadata))
	 (description (alist-get 'description metadata))
	 (path-chain (alist-get 'path-chain metadata))
	 (ancestors-names (alist-get 'ancestors-names metadata))
	 (template (if layout
		       (nori-site-load-template layout)
		     (nori-site-load-template (car path-chain))))
	 (date (alist-get 'date metadata))
	 (tags (alist-get 'tags metadata))
	 (wordcount (alist-get 'wordcount metadata))
	 (menu-name (alist-get 'menu-name metadata))
	 (menu-html (nori-site-build-menu menu-name))
					;(sidebar-html (nori-site-build-sidebar metadata))
	 (replacements
	  `(("{{TITLE}}" . ,title)
	    ("{{CONTENTS}}" . ,html)
	    ("{{LANGUAGE_CODE}}" . ,nori-site-language-code)
	    ("{{CANONICAL_URL}}" . ,(concat nori-site-base-url (string-join path-chain "/") "/" (unless file-name "index")))
					;("{{URL}}" . ,(concat "/" (string-join path-chain "/") "/" file-name))
	    ("{{URL}}" . ,url)
	    ("{{SECTION_TITLE}}" . ,(car ancestors-names))
	    ("{{SECTION_URL}}" . ,(concat "/" (string-join path-chain "/")))
	    ("{{DATE}}" . ,date)
	    ("{{DATE_HUMAN}}" . ,(when date (format-time-string "%B %d, %Y" (date-to-time date))))
	    ("{{WORDCOUNT}}" . ,(int-to-string wordcount))
	    ;; Average reading speed is about 238 words/min
	    ("{{READING_TIME}}" . ,(int-to-string (/ wordcount 238)))
	    ("{{MENU_HTML}}" . ,menu-html)
	    ("{{ARCHIVE}}" . ,archive-html)
					;("{{SIDEBAR_HTML}}" . ,sidebar-html)
	    ("{{CURRENT_YEAR}}" . ,(format-time-string "%Y"))
	    ("{{FAVICON}}" . ,nori-site-favicon)
	    ("{{SITE_NAME}}" . ,nori-site-title)
	    ("{{FEATURED_IMAGE}}" . ,(concat nori-site-base-url f-image))

	    ("{{SITE_LOGO_URL}}" . ,nori-site-logo)
	    ("{{SITE_TITLE}}" . ,nori-site-title)

(defun nori-site-export-subtree-to-html (buffer begin)
  "Export a subtree contents to HTML with debugging."
  (with-current-buffer buffer
    (save-excursion ;saves buffer selection and point
      (save-restriction ;saves narrowing state
	(goto-char begin)
					;(message "DEBUG: At point: %s" (buffer-substring (line-beginning-position) (line-end-position)))
	(let ((exported-html nil)
	      (export-buffer
	       (org-html-export-as-html
		nil       ; async
		t         ; subtreep
		nil       ; visible-only
		t         ; body-only
		'(:with-toc nil
			    :section-numbers nil)       ; ext-plist default nil
		)))
	  (unwind-protect
              (setq exported-html
		    (with-current-buffer export-buffer
		      (buffer-string)))
	    (kill-buffer export-buffer))
					;(message "DEBUG: Exported html length: %d" (length exported-html))
	  ;; Fix image links, they need and extra / src="img -> src="\img
	  (setq exported-html (replace-regexp-in-string "src=\"img" "src=\"\\\\img" exported-html))
	  
	  exported-html
	  )))))

(defun nori-site-render-template (template-string replacements)
  "Replace PLACEHOLDERS in TEMPLATE_STRING using REPLACEMENTS alist.
REPLACEMENTS is a list of (PLACEHOLDER . VALUE) pairs."
  (let ((result template-string))
    (dolist (pair replacements)
      (setq result (replace-regexp-in-string
		    (car pair)
		    (or (cdr pair) "")
		    result
		    t t))) ; case-sensitive, literal match
    result))

(defun nori-site-build-menu (menu-name)
  "Build a flat HTML menu.
CURRENT-FILE is the EXPORT_FILE_NAME of the page being rendered,
used to add the active class."
  (let ((menu-items '(("Home" . "/")
		      ("Posts" . "/post/")
		      ("Fiction" . "/alphabet_superset/")
		      ("Microblog" . "/microblog/")))
	(html "  <nav class=\"site-nav\">\n    <ul>\n")
	)

    ;; Build the HTML string
    (dolist (item menu-items)
      (let* ((title (car item))
	     (menu-url (cdr item))
	     (is-active (and menu-name (string= menu-name title)))
	     (attrs (if is-active
                        "aria-current=\"page\" class=\"active\""
                      "")))
        (setq html
              (concat html
                      (format "    <li>\n      <a %s href=\"%s\">%s</a>\n    </li>\n"
                              attrs menu-url title)))))
    (setq html (concat html "    </ul>\n  </nav>"))
    html))

(defun nori-site-gather-items (parent-section-name)
  "Gather all items with a given PARENT-SECTION-NAME into an alist of the same form as nori-site-entry-index."
  ;; Gather items
  ;; Parse the org-source if not parsed yet
  (unless nori-site-entry-index
    (nori-site-build-entry-index nori-site-source-org-filename))
  (let ((items (copy-alist nori-site-entry-index)))
    (setq items (cl-remove-if
		 (lambda (item)
		   (let* ((metadata (cdr item))
			  (path-chain (cdr (assoc 'path-chain metadata)))
			  (parent (car path-chain))
			  (todo (cdr (assoc 'todo metadata)))
			  (date (cdr (assoc 'date metadata)))
			  (file-name (cdr (assoc 'file-name metadata)))
			  )

		     (or (not date) (not file-name) (not (equal parent-section-name parent)) (and todo (string= todo "TODO")))))
		 items))
    (message "%s count %d" parent-section-name (length items))
    ;; Sort by date descending
    (sort items (lambda (a b) (string> (alist-get 'date (cdr a)) (alist-get 'date (cdr b)))))
    items))

(defun nori-site-generate-archive-sidebar (posts)
  "Generate the HTML for the year archive sidebar using an alist of sorted POSTS"
  ;; Create the archive html
  (let* ((html "<div class=\"year-archive\">")
         (prev-year nil)
         (first-group t))
    
    (dolist (post posts)
      (let* ((meta (cdr post))
	     (date (date-to-time (alist-get 'date meta)))
             (year-str (format-time-string "%Y" date))
             (date-display (format-time-string "%d %b" date))
             (title (alist-get 'title meta))
             (file-name (alist-get 'file-name meta))
	     (path-chain (alist-get 'path-chain meta))
	     (url (concat "/" (string-join path-chain "/") "/" file-name)))
	;(message "%s" title)
        ;; Check if year changed
        (if (not (string= year-str prev-year))
            ;; New Year Group
            (progn
	      ;; Close previous group if not the first one
	      (unless first-group
                (setq html (concat html "    </ul>\n  </details>\n")))
	      
	      ;; Open new group
	      (setq html (concat html
                                 "  <details class=\"year-panel\">\n"
                                 "    <summary class=\"year-header\">" year-str "</summary>\n"
                                 "    <ul class=\"post-list\">\n"))
	      
	      (setq prev-year year-str)
	      (setq first-group nil))
          ;; Same year, just continue
	  
	  )
        
        ;; Add list item
        (setq html (concat html
                           "    <li class=\"post-item\">\n"
                           "      <a href=\"" url "\">" date-display " – " title "</a>\n"
                           "    </li>\n"))))
    
    ;; Close the final group
    (setq html (concat html "    </ul>\n  </details>\n</div>"))
    html))

(defun nori-site-generate-tag-list (tags)
  "Generate the html for the tag list from a TAGS list for a given item."
  (let* ((html "")

      (let* ((url-tag (replace-regexp-in-string " " "-" tag))
	     (url (concat "/" "tags" "/" url-tag)))
	;(message "tag is %s and url is %s" tag url)
	(setq html (concat html "<li><a href=\"" url "\">" tag "</a></li>\n"))))
    html))

(defun paginate-list (items page-size)
  "Split an ordered alist ITEMS into pages of PAGE-SIZE.
Return a list of pages, each element is a plist:
(:number N :total N :items LIST :url STRING :prev URL/nil :next URL/nil ...)."
  (let* ((total-pages (ceiling (/ (length items) page-size)))
	 pages
	 start-index)
    (dotimes (i total-pages)
      (setq start-index (* i page-size))
      (push
       (list :number (1+ i)
	     :total total-pages
	     :items (seq-subseq items start-index (min (+ start-index page-size) (length items)))
	     :url (format "/page/%d/" (1+ i)))
       pages))
    ;; Reverse because we pushed in reverse order
    (setq pages (nreverse pages))

    ;; Add prev/next/first/last references
    (cl-loop for page in pages
	     for idx from 0
	     do
	     (setf (plist-get page :prev)
		   (when (> idx 0)
		     (plist-get (nth (- idx 1) pages :url))))
	     (setf (plist-get page :next)
		   (when (< idx (1- (length pages)))
		     (plist-get (nth (+ idx 1) pages :url))))
	     (setf (plist-get page :first)
		   (plist-get (car pages) :url))
	     (setf (plist-get page :last)
		   (plist-get (car (last pages)) :url)))
    pages))

(defun nori-site-process-headline (headline buffer current-path)
  (let* ((metadata (nori-site-parse-headline headline buffer))
	 (section
	  (alist-get 'section-name metadata))
	 (file
	  (alist-get 'file-name metadata))
	 (begin
	  (alist-get 'begin metadata))
	 (todo-state
	  (alist-get 'todo metadata))
	 )
    (if (and todo-state (string= todo-state "TODO"))
	nil
    
    (cond
     ;; section
     (section ;check if it's a section
      (let ((section-dir (expand-file-name (nori-site-concat-to-dir current-path section))))
	(unless (file-exists-p section-dir)
	  (make-directory section-dir t))
	(dolist (child
		 (org-element-contents headline)) ;only go through the children of the headline, not the full tree
	  (when (eq (org-element-type child) 'headline)
	    (nori-site-process-headline child buffer section-dir)))
	))
     ;; file
     (file
      (let* ((file-dir (expand-file-name (nori-site-concat-to-dir current-path file)))
	     (html (nori-site-export-subtree-to-html buffer begin))
	     (filename (if (or (string= file "_index")(string= file "index"))
			   (expand-file-name "index.html" current-path)
			 (expand-file-name "index.html" file-dir))))
	(unless (or (file-exists-p file-dir) (string= file "index") (string= file "_index"))
	  (make-directory file-dir t))
	(let ((page (nori-site-generate-page metadata html)))
          (with-temp-file filename
            (insert page)))
	(message "Published: %s -> %s" (alist-get 'title metadata) filename)
	))
     ;; else do nothing
     (t nil)))))

(defun nori-site-parse-headline (headline buffer)
  (with-current-buffer buffer
    (save-excursion
      (goto-char (org-element-property :begin headline))
      (let* ((begin (org-element-property :begin headline))
	     (contents-begin (org-element-property :contents-begin headline))
	     (contents-end (org-element-property :contents-end headline))
	     (title (substring-no-properties
		     (org-element-interpret-data
		      (org-element-property :title headline))))
	     (props (org-entry-properties begin 'all))
	     (export-file-name (alist-get
				"EXPORT_FILE_NAME"
				props nil nil #'string=))
	     (date (alist-get
		    "EXPORT_DATE"
		    props nil nil #'string=))
             (categories (org-entry-get
			  begin
			  "EXPORT_CATEGORIES"
			  :inherit))
	     (section-name (alist-get
			    "EXPORT_SECTION"
			    props nil nil #'string=))
	     (layout (alist-get "EXPORT_LAYOUT"
				props nil nil #'string=))
	     (f-image (alist-get "EXPORT_FEATURED_IMAGE"
				props nil nil #'string=))
	     (f-alt (alist-get "EXPORT_FEATURED_ALT"
				props nil nil #'string=))
	     (description (alist-get "DESCRIPTION"
				props nil nil #'string=))
	     (menu-name (alist-get "MENU_NAME"
				   props nil nil #'string=))
	     (todo (org-entry-get
                    begin
                    "TODO"
                    :inherit))
	     (tags (org-entry-get
		    begin
		    "ALLTAGS"))
