(defun nori-site-export-subtree-to-html (buffer org-file begin &optional debug)
  "Export a subtree contents to HTML with debugging."
  (with-current-buffer buffer
    (save-excursion ;saves buffer selection and point
      (save-restriction ;saves narrowing state
	(goto-char begin)
	(when debug
	  (message "DEBUG: At point: %s" (buffer-substring (line-beginning-position) (line-end-position))))
	(let ((exported-html nil)
	      (export-buffer
	       (org-html-export-as-html
		nil       ; async
		t         ; subtreep
		nil       ; visible-only
		t         ; body-only
		nil       ; ext-plist
		)))
	  (unwind-protect
              (setq exported-html
		    (with-current-buffer export-buffer
		      (buffer-string)))
	    (kill-buffer export-buffer))
	  (when debug
            (message "DEBUG: Exported html length: %d" (length exported-html)))

	  exported-html
	  )))))

(defun nori-site-get-all-posts-from-org (org-file &optional debug)
  "Extract all posts from an org file."
  (with-current-buffer (find-file-noselect org-file)
    (let ((posts '())
	  ;; Parse the buffer once
	  (ast (org-element-parse-buffer)))

      (org-element-map ast 'headline
	(lambda (headline)
	  (when (= (org-element-property :level headline) 2)
	    (let* ((begin (org-element-property :begin headline))
		   (contents-begin (org-element-property :contents-begin headline))
                   (contents-end (org-element-property :contents-end headline))
		   (parent (org-element-lineage headline '(headline)))
		   (section-title (if parent
				      (substring-no-properties
				       (org-element-interpret-data
					(org-element-property :title parent)))
				    "Unknown"))
		   (title (substring-no-properties
			   (org-element-interpret-data
			    (org-element-property :title headline))))
		   (parent-props (org-entry-properties parent))
		   (props (org-entry-properties begin 'all))
		   (export-section (alist-get
				    "EXPORT_HUGO_SECTION"
				    parent-props nil nil #'string=))
		   (export-file-name (alist-get
                                      "EXPORT_FILE_NAME"
                                      props nil nil #'string=))
		   (date (alist-get
			  "EXPORT_DATE"
			  props nil nil #'string=))
		   (tags (org-entry-get
                          begin
                          "EXPORT_HUGO_TAGS"
                          :inherit))
                   (categories (org-entry-get
                                begin
                                "EXPORT_HUGO_CATEGORIES"
                                :inherit))
		   (todo (org-entry-get
                          begin
                          "TODO"
                          :inherit))
		   (html (nori-site-export-subtree-to-html (current-buffer) org-file begin nil)))
	      (when debug
		(message "DEBUG: Props %s" props)
		(message "DEBUG: Parent props %s" parent-props))
	      (push `((title . ,title)
		      (section-title . ,section-title)
		      (file-name . ,export-file-name)
		      (export-section . ,export-section)
		      (todo . ,todo)
		      (date . ,date)
		      (tags . ,tags)
		      (categories . ,categories)
		      (begin . ,begin)
		      (contents-begin . ,contents-begin)
		      (contents-end . ,contents-end)
		      (html . ,html)
		      )
		    posts)
	      ))))
      (when debug (message "DEBUG: Final posts list length: %d" (length posts)))
      (nreverse posts))
    ))

(defun nori-site-make-site (org-file &optional debug)
  "Generate the full site from the ORG-FILE"
  (let ((publish-dir (expand-file-name nori-site-publish-directory))
	(posts '()))
    ;; Ensure publish directory exists
    (unless (file-exists-p publish-dir)
      (make-directory publish-dir t))
    ;; Change to org-source buffer
    (with-current-buffer (find-file-noselect org-file)
      ;; Parse the org-source buffer
      (let ((ast (org-element-parse-buffer)))
	(when debug
	  (message "Got the source file ast"))
	;; Process each headline
	(org-element-map ast 'headline
	  (lambda (headline)
	    ;; Parse headline of posts for metadata and content
	    (when (= (org-element-property :level headline) 1)
	      (let* ((section (nori-site-parse-headline headline)))
		(when (alist-get 'section-name section)
		  (let ((section-dir (expand-file-name (nori-site-concat-to-dir publish-dir (alist-get 'section-name section))))
			)
		    (unless (file-exists-p section-dir)
		      (make-directory section-dir t))
		    (message "DEBUG: Contents end is %s" (alist-get 'contents-end section))
		    ))))
		     
	    (when (= (org-element-property :level headline) 2)
	      (let* ((post (nori-site-parse-headline headline))
		     (section-dir (nori-site-concat-to-dir publish-dir (alist-get 'export-section post)))
		     (output-file (expand-file-name (concat (alist-get 'file-name post) ".html") section-dir)))
		
		(when debug
		  (message "Title is %s" (alist-get 'title post))
		  (message "Publish dir is %s" publish-dir)
		  (message "Filename is %s" output-file))
		
		(push post posts)
		))))))
					;(nreverse posts)
    (when debug
      (message "Number of posts is %s" (length posts)))
    posts)
  )

;; Render all posts to HTML
(defun nori-site-generate-all-posts ()
  "Parse source file and generate HTML for each post."
  (let ((posts (nori-site-site-parse-posts))
        (publish-dir (expand-file-name nori-site-publish-directory)))

    ;; Ensure publish directory exists
    (unless (file-exists-p publish-dir)
      (make-directory publish-dir t))

    ;; Process each post
    (dolist (post posts)
      (let* ((title (cdr (assoc 'title post)))
             (slug (nori-site-post-slug post))
             (date (nori-site-post-date post))
             (content (cdr (assoc 'content post)))
             (output-file (expand-file-name (concat slug ".html") publish-dir))
             (url (concat nori-site-base-url slug "/"))

             ;; Convert Org content to HTML
             (html-content (nori-site-org-to-html content)))

        ;; Render using template
        (let ((html (nori-site-generate-page title html-content url)))
          (with-temp-file output-file
            (insert html)))

        (message "Published: %s -> %s" title output-file)))

    ;; Also generate index/home page
    (nori-site-generate-index posts)))

;; Convert Org text to HTML (using org-mode export)
(defun nori-site-org-to-html (org-text)
  "Convert ORG-TEXT to HTML using org-mode export."
  (let ((html nil))
    (with-temp-buffer
      (insert org-text)
      (setq html (org-html-export-to-html))
      (goto-char (point-min))
      ;; Extract just the body content, not full document
      (when (search-forward "<body>" nil t)
        (let ((start (point)))
          (when (search-forward "</body>" nil t)
            (setq html (buffer-substring start (1- (point))))))))
    html))

;; Helper to load a template file
(defun nori-site-load-template (filename)
  "Read the content of TEMPLATE FILENAME."
  (with-temp-buffer
    (insert-file-contents (nori-site-concat-to-dir
			   (nori-site-concat-to-dir nori-site-main-dir "templates/") filename)
			  (buffer-string))))

;; Helper to replace placeholders
(defun nori-site-render-template (template-string replacements)
  "Replace PLACEHOLDERS in TEMPLATE_STRING using REPLACEMENTS alist.
REPLACEMENTS is a list of (PLACEHOLDER . VALUE) pairs."
  (let ((result template-string))
    (dolist (pair replacements)
      (setq result (replace-regexp-in-string
		    (car pair)
		    (cdr pair)
		    result
		    t t))))) ;literal, case-sensitive match

(defun nori-site-generate-page (title content url)
  "Generate a page using the template."
  (let* ((template (nori-site-load-template "page.html"))
	 (replacements
	  '(("{{TITLE}}" . ,title)
	    ("{{CONTENT}}" . ,content)
            ("{{URL}}" . ,(concat nori-site-base-url url))
            ("{{LANGUAGE_CODE}}" . ,nori-site-language-code)
            ("{{FAVICON}}" . ,nori-site-favicon)
            ;; You can generate complex blocks like Open Graph tags separately
            ("{{META_DESCRIPTION}}" . ,(format "<meta name=\"description\" content=\"%s\">"
                                               (or (get-text-property 0 'description content) nori-site-description)))
            ("{{OPEN_GRAPH_TAGS}}" . ,(nori-site-generate-og-tags title url))
            ("{{HEADER_CONTENT}}" . ,(nori-site-generate-header))
            ("{{FOOTER_CONTENT}}" . ,(nori-site-generate-footer)))))

    (nori-site-render-template template replacements)))
